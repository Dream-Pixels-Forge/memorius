```python
def invalidate_edge(
    conn: sqlite3.Connection,
    source_id: str,
    target_id: str,
    relation: str = "related",
):
    """Soft-delete an edge by setting tinvalid to now.

    The edge remains in the table for historical queries but is excluded
    from active retrieval (WHERE tinvalid IS NULL).
    """
    now = datetime.now(timezone.utc).isoformat()
    conn.execute(
        "UPDATE memory_graph SET tinvalid = ? "
        "WHERE source_id = ? AND target_id = ? AND relation = ? AND tinvalid IS NULL",
        (now, source_id, target_id, relation),
    )
    conn.commit()
```

```python
def link_memories(
    conn: sqlite3.Connection,
    source_id: str,
    target_id: str,
    weight: float = 1.0,
    relation: str = "related",
):
    """Create or update a bidirectional link between two memories.

    If an active edge already exists with the same relation, the old edge
    is invalidated (temporal soft-delete) and a fresh edge is created.
    """
    now = datetime.now(timezone.utc).isoformat()
    try:
        # Invalidate any existing active edge with same relation
        conn.execute(
            "UPDATE memory_graph SET tinvalid = ? "
            "WHERE source_id = ? AND target_id = ? AND relation = ? AND tinvalid IS NULL",
            (now, source_id, target_id, relation),
        )
        conn.execute(
            "UPDATE memory_graph SET tinvalid = ? "
            "WHERE source_id = ? AND target_id = ? AND relation = ? AND tinvalid IS NULL",
            (now, target_id, source_id, relation),
        )
        # Insert fresh edges
        conn.execute(
            "INSERT OR IGNORE INTO memory_graph (source_id, target_id, weight, relation, created_at, tvalid) "
            "VALUES (?, ?, ?, ?, ?, ?)",
            (source_id, target_id, weight, relation, now, now),
        )
        conn.execute(
            "INSERT OR IGNORE INTO memory_graph (source_id, target_id, weight, relation, created_at, tvalid) "
            "VALUES (?, ?, ?, ?, ?, ?)",
            (target_id, source_id, weight, relation, now, now),
        )
        conn.commit()
    except sqlite3.OperationalError:
        init_graph_schema(conn)
        link_memories(conn, source_id, target_id, weight, relation)
```

```python
def get_edges_at_time(
    conn: sqlite3.Connection,
    memory_id: str,
    at_time: str | None = None,
) -> list[dict[str, Any]]:
    """Get edges active at a given point in time.

    Args:
        memory_id: Source memory to query from.
        at_time: ISO timestamp. None means "right now" (only active edges).
    """
    if at_time is None:
        rows = conn.execute(
            """SELECT source_id, target_id, weight, relation, tvalid, tinvalid
               FROM memory_graph
               WHERE source_id = ? AND tinvalid IS NULL
               ORDER BY weight DESC""",
            (memory_id,),
        ).fetchall()
    else:
        rows = conn.execute(
            """SELECT source_id, target_id, weight, relation, tvalid, tinvalid
               FROM memory_graph
               WHERE source_id = ?
               AND tvalid <= ?
               AND (tinvalid IS NULL OR tinvalid > ?)
               ORDER BY weight DESC""",
            (memory_id, at_time, at_time),
        ).fetchall()
    return [dict(r) for r in rows]


def get_active_edge_count(conn: sqlite3.Connection, memory_id: str) -> int:
    """Count active (non-invalidated) edges for a memory."""
    row = conn.execute(
        "SELECT COUNT(*) FROM memory_graph WHERE source_id = ? AND tinvalid IS NULL",
        (memory_id,),
    ).fetchone()
    return row[0] if row else 0
```

```python
# Add to memorius/temporal.py

# Heat score tiers
TIER_HOT = "hot"       # heat_score >= 0.7
TIER_WARM = "warm"     # 0.3 <= heat_score < 0.7
TIER_COLD = "cold"     # heat_score < 0.3

def calculate_heat_score(
    created_at: str,
    last_accessed: str | None = None,
    access_count: int = 0,
) -> float:
    """Calculate heat score combining recency, frequency, and age.

    Returns a value between 0.0 (cold) and 1.0 (hot).
    """
    now = datetime.now(timezone.utc)

    # Recency component (0-1, higher = more recent)
    recency = 0.5  # default for no access
    if last_accessed:
        try:
            accessed = datetime.fromisoformat(last_accessed.replace("Z", "+00:00"))
            days_since = max((now - accessed).total_seconds() / 86400, 0)
            recency = 1.0 / (1.0 + days_since * 0.1)  # slow decay
        except (ValueError, AttributeError):
            pass

    # Frequency component (0-1, logarithmic)
    import math
    frequency = min(math.log(access_count + 1, 2) / 5.0, 1.0)

    # Freshness component (0-1, newer = higher)
    freshness = 0.5
    try:
        created = datetime.fromisoformat(created_at.replace("Z", "+00:00"))
        days_old = max((now - created).total_seconds() / 86400, 0)
        freshness = 1.0 / (1.0 + days_old * 0.05)  # very slow decay
    except (ValueError, AttributeError):
        pass

    # Weighted combination
    score = recency * 0.4 + frequency * 0.3 + freshness * 0.3
    return round(max(0.0, min(1.0, score)), 4)


def classify_tier(heat_score: float) -> str:
    """Classify a heat score into a tier label."""
    if heat_score >= 0.7:
        return TIER_HOT
    elif heat_score >= 0.3:
        return TIER_WARM
    else:
        return TIER_COLD
```

```python
WEIGHT_HEAT = 0.15  # heat score weight in final ranking

def calculate_search_score_with_heat(
    semantic_similarity: float,
    decay_score: float,
    access_count: int = 0,
    heat_score: float = 0.5,
    semantic_weight: float = WEIGHT_SEMANTIC,
    decay_weight: float = WEIGHT_TEMPORAL,
    access_weight: float = WEIGHT_ACCESS,
    heat_weight: float = WEIGHT_HEAT,
) -> float:
    """Calculate final search score including heat-tier boosting."""
    reinforcement = math.log(access_count + 1, REINFORCEMENT_LOG_BASE) + 1.0
    reinforcement = min(reinforcement, 3.0) / 3.0

    score = (
        semantic_similarity * semantic_weight
        + decay_score * decay_weight
        + reinforcement * access_weight
        + heat_score * heat_weight
    )
    return round(score, 4)
```

```python
def _check_for_duplicates(
    self, content: str, vault: str, shelf: str, threshold: float = 0.92
) -> dict[str, Any] | None:
    """Check if content is a near-duplicate of an existing memory."""
    # Search for similar content in same vault/shelf
    results = self._vector.search(content, vault=vault, shelf=shelf, n_results=3)
    for mem in results:
        if mem.content.strip() == content.strip():
            return {"id": mem.id, "similarity": 1.0}
        # Cosine similarity check (embedding already computed)
        if mem.vector and hasattr(self, '_embed'):
            new_vec = self._embed.embed(content)
            from memorius.utils import cosine_similarity
            sim = cosine_similarity(new_vec, mem.vector)
            if sim >= threshold:
                return {"id": mem.id, "similarity": sim}
    return None
```

```python
class ConsolidationDaemon:
    """Background thread that runs consolidation on a schedule.

    Modeled after Claude Code's Auto Dream:
    1. Orient — scan memory directory
    2. Gather signals — identify duplicates, contradictions, stale entries
    3. Consolidate — merge, dedup, resolve conflicts
    4. Prune — archive low-value memories
    """

    def __init__(self, engine, config: dict[str, Any]):
        self._engine = engine
        self._config = config
        self._interval = config.get("interval_hours", 24) * 3600
        self._threshold = config.get("similarity_threshold", 0.80)
        self._min_sessions = config.get("min_sessions", 5)
        self._max_per_run = config.get("max_memories_per_run", 500)
        self._running = False
        self._thread = None
        self._last_run = None

    def start(self):
        """Start the background daemon."""
        if self._running:
            return
        self._running = True
        self._thread = threading.Thread(target=self._loop, daemon=True, name="memorius-consolidation")
        self._thread.start()
        logger.info("Consolidation daemon started")

    def stop(self):
        """Stop the background daemon."""
        self._running = False
        if self._thread:
            self._thread.join(timeout=5)

    def is_alive(self) -> bool:
        return self._thread is not None and self._thread.is_alive()

    def run_once(self, dry_run: bool = False) -> ConsolidationResult:
        """Run a single consolidation pass (the "dream" cycle)."""
        logger.info("Starting consolidation dream cycle")
        result = consolidate(
            self._engine,
            similarity_threshold=self._threshold,
            dry_run=dry_run,
        )
        self._last_run = datetime.now(timezone.utc).isoformat()
        return result

    def _loop(self):
        """Background loop — sleep then consolidate."""
        while self._running:
            time.sleep(self._interval)
            if not self._running:
                break
            try:
                self.run_once()
            except Exception as e:
                logger.error(f"Consolidation daemon error: {e}")
```

```python
def bm25_search(
    self,
    query: str,
    vault: str | None = None,
    limit: int = 20,
) -> list[dict[str, Any]]:
    """BM25 keyword search via FTS5.

    Returns list of dicts with 'id' and 'rank' keys, ordered by relevance.
    """
    try:
        sql = "SELECT id, rank FROM memory_fts WHERE memory_fts MATCH ?"
        params: list[Any] = [query]
        if vault:
            sql += " AND vault = ?"
            params.append(vault)
        sql += " ORDER BY rank LIMIT ?"
        params.append(limit)
        rows = self._conn().execute(sql, params).fetchall()
        return [dict(r) for r in rows]
    except sqlite3.OperationalError:
        return []
```

```python
def calculate_hybrid_score(
    vector_score: float,
    bm25_score: float,
    vector_weight: float = 0.6,
    bm25_weight: float = 0.4,
) -> float:
```
